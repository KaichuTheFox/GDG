cmake_minimum_required(VERSION 3.20)
project(ZCompiler VERSION 0.2.0 LANGUAGES CXX)
include(CTest)
option(ZLANG_WARNINGS_AS_ERRORS "Treat warnings as errors" OFF)
add_library(zlang_core src/Lexer.cpp src/Parser.cpp src/Semantic.cpp src/CodeGen.cpp src/Diagnostic.cpp src/Process.cpp)
target_include_directories(zlang_core PUBLIC include)
target_compile_features(zlang_core PUBLIC cxx_std_17)
if(MSVC)
 target_compile_options(zlang_core PRIVATE /W4 /utf-8 $<$<BOOL:${ZLANG_WARNINGS_AS_ERRORS}>:/WX>)
else()
 target_compile_options(zlang_core PRIVATE -Wall -Wextra -Wpedantic $<$<BOOL:${ZLANG_WARNINGS_AS_ERRORS}>:-Werror>)
endif()
add_executable(zlang src/main.cpp)
target_link_libraries(zlang PRIVATE zlang_core)
if(BUILD_TESTING)
 add_executable(zlang_unit tests/unit.cpp)
 target_link_libraries(zlang_unit PRIVATE zlang_core)
 add_test(NAME unit COMMAND zlang_unit)
 add_test(NAME compatibility COMMAND ${CMAKE_COMMAND} -DZLANG=$<TARGET_FILE:zlang> -DSOURCE=${CMAKE_SOURCE_DIR}/examples/compatibility.z -DEXPECTED=${CMAKE_SOURCE_DIR}/examples/compatibility.stdout -DWORK=${CMAKE_BINARY_DIR}/it -P ${CMAKE_SOURCE_DIR}/tests/run_test.cmake)
endif()
