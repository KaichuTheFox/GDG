# Grammar
`program = { function | terminator }; function = identifier "(" [identifier {"," identifier}] ")" block; block = "{" {statement | terminator} "}"; statement = declaration | assignment | print | if | while | for | return | expression; declaration = ("box"|"line"|"bool"|"cpx") identifier "=" expression; assignment = identifier "=" expression; print = "print" ("(" [arguments] ")" | arguments); if = "if" "(" expression ")" block {"else" "if" "(" expression ")" block} ["else" block]; while = "while" "(" expression ")" block; for = "for" identifier "in" "+" "(" expression ")" block; return = "return" [expression]; arguments = expression {"," expression}; expression uses precedence: call, unary, multiply, add, compare, equality. terminator = newline | ";".`
